cmake_minimum_required(VERSION 3.22)
project(live_whisper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(WHISPER_CPP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../vendor/whisper.cpp")
if(NOT EXISTS "${WHISPER_CPP_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Missing whisper.cpp: clone ggml-org/whisper.cpp into android/app/vendor/whisper.cpp, see README")
endif()
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${WHISPER_CPP_DIR}" "${CMAKE_BINARY_DIR}/whisper")
add_library(live_whisper SHARED live_whisper.cpp)
target_link_libraries(live_whisper PRIVATE whisper log)
target_include_directories(live_whisper PRIVATE "${WHISPER_CPP_DIR}/include")
